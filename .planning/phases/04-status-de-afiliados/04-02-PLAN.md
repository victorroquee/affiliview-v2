   function daysAgo(iso: string): number {
     const then = new Date(iso + "T00:00:00Z").getTime();
     const now  = Date.now();
     return Math.floor((now - then) / 86400000);
   }

   function formatDaysAgo(iso: string): string {
     const d = daysAgo(iso);
     if (d === 0) return "hoje";
     if (d === 1) return "ontem";
     return `${d} dias atrás`;
