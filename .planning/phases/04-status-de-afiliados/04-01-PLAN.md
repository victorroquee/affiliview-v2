     } else {
       assigned = "Inativo";
     }

4. **In `computeAffiliateRankings` function** — After building `affData` map (after line 201) and before the rankings loop, compute `lastFrontSaleDate` per affiliate by scanning ALL `payRows` (not just window-filtered) for `upsellNo === 0`:
   ```typescript
   // Track last front sale date per affiliate across ALL data
   const lastSaleMap = new Map<string, string>();
   for (const t of payRows) {
     if (t.upsellNo !== 0) continue;
     const name = t.affiliate.trim();
     if (!name) continue;
     const dateKey = t.date.toISOString().split("T")[0]!;
     const prev = lastSaleMap.get(name);
     if (!prev || dateKey > prev) {
       lastSaleMap.set(name, dateKey);
     }
   }
   ```

5. **Line 230-236** — Add `lastFrontSaleDate` to the ranking info object:
   ```typescript
   rankings.set(name, {
     ranking: assigned,
     days,
     frontSalesInWindow: data.frontSales,
     windowStart: dateKeys[0]!,
     windowEnd: dateKeys[6]!,
     lastFrontSaleDate: lastSaleMap.get(name) ?? null,
   });
   ```

6. **Also handle affiliates NOT in the 7-day window** — After the main rankings loop (after line 237), iterate `lastSaleMap` to find affiliates who have historical sales but zero in the window. For each name in `lastSaleMap` NOT already in `rankings`, add them as Inativo:

   // Add affiliates with historical sales but 0 in window as Inativo
   for (const [name, lastDate] of lastSaleMap) {

     const emptyDays: AffiliateDayDetail[] = dateKeys.map((dk) => ({
       date: dk, gross: 0, frontSales: 0, t1: false, t2: false, t3: false,
     }));
     rankings.set(name, {
       ranking: "Inativo",
       days: emptyDays,
       frontSalesInWindow: 0,
       windowStart: dateKeys[0]!,
       windowEnd: dateKeys[6]!,
       lastFrontSaleDate: lastDate,
     });
   }
